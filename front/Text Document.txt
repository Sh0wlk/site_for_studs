    <script>
        const API_URL = "http://127.0.0";
        let authMode = "login";
        let pendingUser = null;
        let currentUser = null;

        const ROLE_NAMES = {
            guest: "Гость (Не подтвержден)",
            student: "Студент",
            teacher: "Преподаватель",
            deanery: "Деканат",
            admin: "Администратор"
        };

        function setAuthTab(mode) {
            authMode = mode;
            document.getElementById("auth-error").classList.add("hidden");
            document.getElementById("field-fullname").classList.toggle("hidden", mode === "login");
            document.getElementById("input-fullname").required = (mode === "register");
            document.getElementById("auth-submit-btn").textContent = mode === "login"
                ? "Войти в систему"
                : "Подать заявку на регистрацию";

            document.getElementById("tab-login").className = mode === "login"
                ? "flex-1 py-2 text-sm font-semibold rounded-lg bg-white shadow text-slate-900 transition"
                : "flex-1 py-2 text-sm font-semibold rounded-lg text-slate-500 transition";

            document.getElementById("tab-register").className = mode === "register"
                ? "flex-1 py-2 text-sm font-semibold rounded-lg bg-white shadow text-slate-900 transition"
                : "flex-1 py-2 text-sm font-semibold rounded-lg text-slate-500 transition";
        }

        async function submitAuth(e) {
            e.preventDefault();
            const errBox = document.getElementById("auth-error");
            errBox.classList.add("hidden");

            const email = document.getElementById("input-email").value.trim();
            const password = document.getElementById("input-password").value;
            const full_name = document.getElementById("input-fullname").value.trim();

            const payload = authMode === "login"
                ? { email, password }
                : { full_name, email, password };

            try {
                const res = await fetch(`${API_URL}/${authMode}`, {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify(payload)
                });
                const data = await res.json();

                if (!res.ok) {
                    errBox.textContent = data.detail || "Ошибка авторизации";
                    errBox.classList.remove("hidden");
                    return;
                }

                if (authMode === "register") {
                    localStorage.setItem("portal_token", data.token);
                    await checkStatus();
                } else {
                    if (["teacher", "deanery", "admin"].includes(data.user.role)) {
                        pendingUser = data.user;
                        document.getElementById("modal-2fa").classList.remove("hidden");
                    } else {
                        localStorage.setItem("portal_token", data.user.api_token);
                        renderUserState(data.user);
                    }
                }
            } catch (err) {
                errBox.textContent = "Сервер недоступен. Проверьте запуск FastAPI.";
                errBox.classList.remove("hidden");
            }
        }

        function confirm2FA() {
            const code = document.getElementById("input-2fa").value.trim();
            if (code.length !== 6) {
                alert("Введите 6-значный код (например, 123456)");
                return;
            }
            document.getElementById("modal-2fa").classList.add("hidden");
            document.getElementById("input-2fa").value = "";
            localStorage.setItem("portal_token", pendingUser.api_token);
            renderUserState(pendingUser);
        }

        async function checkStatus() {
            const token = localStorage.getItem("portal_token");
            if (!token) return;
            const res = await fetch(`${API_URL}/me`, {
                headers: { "Authorization": `Bearer ${token}` }
            });
            if (res.ok) {
                const user = await res.json();
                renderUserState(user);
            } else {
                logout();
            }
        }

        function renderUserState(user) {
            currentUser = user;
            document.getElementById("auth-screen").classList.add("hidden");

            if (user.role === "guest") {
                document.getElementById("app-screen").classList.add("hidden");
                document.getElementById("guest-screen").classList.remove("hidden");
                document.getElementById("guest-id").textContent = `#${user.id}`;
                document.getElementById("guest-name").textContent = user.full_name;
                return;
            }

            document.getElementById("guest-screen").classList.add("hidden");
            document.getElementById("app-screen").classList.remove("hidden");

            document.getElementById("header-name").textContent = user.full_name;
            const groupLabel = user.group_name ? ` • Группа ${user.group_name}` : "";
            document.getElementById("header-role").textContent = `${ROLE_NAMES[user.role]}${groupLabel}`;

            const isStaff = ["deanery", "admin"].includes(user.role);
            document.getElementById("nav-deanery").classList.toggle("hidden", !isStaff);
            if (isStaff) loadUsersList();
        }

        function openSection(secId) {
            document.querySelectorAll("main section").forEach(el => el.classList.add("hidden"));
            document.getElementById(`sec-${secId}`).classList.remove("hidden");

            document.querySelectorAll(".nav-btn").forEach(btn => {
                const active = btn.dataset.sec === secId;
                btn.classList.toggle("border-indigo-600", active);
                btn.classList.toggle("text-indigo-600", active);
                btn.classList.toggle("border-transparent", !active);
                btn.classList.toggle("text-slate-500", !active);
            });
        }

        async function loadUsersList() {
            const token = localStorage.getItem("portal_token");
            const res = await fetch(`${API_URL}/users`, {
                headers: { "Authorization": `Bearer ${token}` }
            });
            if (!res.ok) return;
            const data = await res.json();

            const tbody = document.getElementById("users-table-body");
            tbody.innerHTML = data.users.map(u => `
                <tr>
                    <td class="py-3 px-2 font-mono text-slate-500">#${u.id}</td>
                    <td class="py-3 px-2 font-medium">${u.full_name}</td>
                    <td class="py-3 px-2 text-slate-500">${u.email}</td>
                    <td class="py-3 px-2">
                        <select id="role-${u.id}" class="border border-slate-300 rounded-lg px-2 py-1 text-xs">
                            <option value="guest" ${u.role === 'guest' ? 'selected' : ''}>Гость</option>
                            <option value="student" ${u.role === 'student' ? 'selected' : ''}>Ученик</option>
                            <option value="teacher" ${u.role === 'teacher' ? 'selected' : ''}>Учитель</option>
                            ${currentUser.role === 'admin' ? `<option value="deanery" ${u.role === 'deanery' ? 'selected' : ''}>Деканат</option>` : ''}
                            ${currentUser.role === 'admin' ? `<option value="admin" ${u.role === 'admin' ? 'selected' : ''}>Админ</option>` : ''}
                        </select>
                    </td>
                    <td class="py-3 px-2">
                        <select id="group-${u.id}" class="border border-slate-300 rounded-lg px-2 py-1 text-xs">
                            <option value="">Без группы</option>
                            ${data.groups.map(g => `<option value="${g.id}" ${u.group_id === g.id ? 'selected' : ''}>${g.name}</option>`).join('')}
                        </select>
                    </td>
                    <td class="py-3 px-2">
                        <button onclick="saveUserRole(${u.id})" class="px-3 py-1 bg-indigo-600 hover:bg-indigo-700 text-white rounded-lg text-xs font-semibold">Сохранить</button>
                    </td>
                </tr>
            `).join('');
        }

        async function saveUserRole(userId) {
            const token = localStorage.getItem("portal_token");
            const role = document.getElementById(`role-${userId}`).value;
            const groupVal = document.getElementById(`group-${userId}`).value;
            const group_id = groupVal ? parseInt(groupVal) : null;

            const res = await fetch(`${API_URL}/users/role`, {
                method: "POST",
                headers: {
                    "Content-Type": "application/json",
                    "Authorization": `Bearer ${token}`
                },
                body: JSON.stringify({ user_id: userId, role, group_id })
            });

            if (res.ok) {
                alert("Роль и группа обновлены!");
                loadUsersList();
            } else {
                const err = await res.json();
                alert(err.detail || "Ошибка сохранения");
            }
        }

        function logout() {
            localStorage.removeItem("portal_token");
            location.reload();
        }

        checkStatus();
    </script>

    <script>
        const API_URL = "http://127.0.0";
        let authMode = "login";
        let pendingUser = null;
        let currentUser = null;
        let allGroups = [];
        let activeTest = null;

        const ROLE_NAMES = {
            guest: "Гость (Не подтвержден)",
            student: "Студент",
            teacher: "Преподаватель",
            deanery: "Деканат",
            admin: "Администратор"
        };

        // --- АВТОРИЗАЦИЯ И РЕГИСТРАЦИЯ ---
        function setAuthTab(mode) {
            authMode = mode;
            document.getElementById("auth-error").classList.add("hidden");
            document.getElementById("field-fullname").classList.toggle("hidden", mode === "login");
            document.getElementById("input-fullname").required = (mode === "register");
            document.getElementById("auth-submit-btn").textContent = mode === "login"
                ? "Войти в систему"
                : "Подать заявку на регистрацию";

            document.getElementById("tab-login").className = mode === "login"
                ? "flex-1 py-2 text-sm font-semibold rounded-lg bg-white shadow text-slate-900 transition"
                : "flex-1 py-2 text-sm font-semibold rounded-lg text-slate-500 transition";

            document.getElementById("tab-register").className = mode === "register"
                ? "flex-1 py-2 text-sm font-semibold rounded-lg bg-white shadow text-slate-900 transition"
                : "flex-1 py-2 text-sm font-semibold rounded-lg text-slate-500 transition";
        }

        async function submitAuth(e) {
            e.preventDefault();
            const errBox = document.getElementById("auth-error");
            errBox.classList.add("hidden");

            const email = document.getElementById("input-email").value.trim();
            const password = document.getElementById("input-password").value;
            const full_name = document.getElementById("input-fullname").value.trim();

            const payload = authMode === "login"
                ? { email, password }
                : { full_name, email, password };

            try {
                const res = await fetch(`${API_URL}/${authMode}`, {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify(payload)
                });
                const data = await res.json();

                if (!res.ok) {
                    errBox.textContent = data.detail || "Ошибка авторизации";
                    errBox.classList.remove("hidden");
                    return;
                }

                if (authMode === "register") {
                    localStorage.setItem("portal_token", data.token);
                    await checkStatus();
                } else if (["teacher", "deanery", "admin"].includes(data.user.role)) {
                    pendingUser = data.user;
                    document.getElementById("modal-2fa").classList.remove("hidden");
                } else {
                    localStorage.setItem("portal_token", data.user.api_token);
                    renderUserState(data.user);
                }
            } catch (err) {
                errBox.textContent = "Сервер недоступен. Проверьте запуск FastAPI.";
                errBox.classList.remove("hidden");
            }
        }

        function confirm2FA() {
            const code = document.getElementById("input-2fa").value.trim();
            if (code.length !== 6) {
                alert("Введите 6-значный код (например, 123456)");
                return;
            }
            document.getElementById("modal-2fa").classList.add("hidden");
            document.getElementById("input-2fa").value = "";
            localStorage.setItem("portal_token", pendingUser.api_token);
            renderUserState(pendingUser);
        }

        async function checkStatus() {
            const token = localStorage.getItem("portal_token");
            if (!token) return;
            const res = await fetch(`${API_URL}/me`, {
                headers: { "Authorization": `Bearer ${token}` }
            });
            if (res.ok) {
                renderUserState(await res.json());
            } else {
                logout();
            }
        }

        function renderUserState(user) {
            currentUser = user;
            document.getElementById("auth-screen").classList.add("hidden");

            if (user.role === "guest") {
                document.getElementById("app-screen").classList.add("hidden");
                document.getElementById("guest-screen").classList.remove("hidden");
                document.getElementById("guest-id").textContent = `#${user.id}`;
                document.getElementById("guest-name").textContent = user.full_name;
                return;
            }

            document.getElementById("guest-screen").classList.add("hidden");
            document.getElementById("app-screen").classList.remove("hidden");
            document.getElementById("header-name").textContent = user.full_name;
            const groupLabel = user.group_name ? ` • Группа ${user.group_name}` : "";
            document.getElementById("header-role").textContent = `${ROLE_NAMES[user.role]}${groupLabel}`;

            const isStaff = ["deanery", "admin"].includes(user.role);
            const isTeacher = ["teacher", "admin"].includes(user.role);

            document.getElementById("nav-deanery").classList.toggle("hidden", !isStaff);
            document.getElementById("news-create-box").classList.toggle("hidden", !isStaff);
            document.getElementById("test-creator-box").classList.toggle("hidden", !isTeacher);
            document.getElementById("ticket-create-box").classList.toggle("hidden", user.role !== "student");

            if (isStaff) loadUsersList();
            else document.getElementById("users-table-body").innerHTML = "";

            openSection("dashboard");
        }

        function openSection(secId) {
            document.querySelectorAll("main section").forEach(el => el.classList.add("hidden"));
            document.getElementById(`sec-${secId}`).classList.remove("hidden");

            document.querySelectorAll(".nav-btn").forEach(btn => {
                const active = btn.dataset.sec === secId;
                btn.classList.toggle("border-indigo-600", active);
                btn.classList.toggle("text-indigo-600", active);
                btn.classList.toggle("border-transparent", !active);
                btn.classList.toggle("text-slate-500", !active);
            });

            if (secId === "dashboard") { loadNews(); loadTeachers(); }
            if (secId === "schedule") loadSchedule();
            if (secId === "tests") { loadTeachers(); loadTests(); }
            if (secId === "tickets") { loadTeachers(); loadTickets(); }
            if (secId === "deanery") loadUsersList();
        }
        // --- НОВОСТИ И ПРЕПОДАВАТЕЛИ ---
        async function loadNews() {
            const token = localStorage.getItem("portal_token");
            const res = await fetch(`${API_URL}/news`, {
                headers: { "Authorization": `Bearer ${token}` }
            });
            const news = await res.json();

            document.getElementById("news-list").innerHTML = news.map(n => `
                <article class="bg-white p-5 rounded-2xl shadow-sm border border-slate-200">
                    <div class="flex items-center justify-between mb-2">
                        <span class="px-2.5 py-0.5 bg-indigo-50 text-indigo-700 rounded-lg text-xs font-semibold">${n.tag}</span>
                        <span class="text-xs text-slate-400">${n.date} • ${n.author_name || 'Портал'}</span>
                    </div>
                    <h3 class="font-bold text-slate-900 text-base">${n.title}</h3>
                    <p class="text-sm text-slate-600 mt-1.5 whitespace-pre-line">${n.content}</p>
                </article>
            `).join('');
        }

        async function publishNews(e) {
            e.preventDefault();
            const token = localStorage.getItem("portal_token");
            const title = document.getElementById("news-title").value.trim();
            const content = document.getElementById("news-content").value.trim();
            const tag = document.getElementById("news-tag").value;

            await fetch(`${API_URL}/news`, {
                method: "POST",
                headers: { "Content-Type": "application/json", "Authorization": `Bearer ${token}` },
                body: JSON.stringify({ title, content, tag })
            });

            document.getElementById("news-title").value = "";
            document.getElementById("news-content").value = "";
            loadNews();
        }

        async function loadTeachers() {
            const token = localStorage.getItem("portal_token");
            const res = await fetch(`${API_URL}/teachers`, {
                headers: { "Authorization": `Bearer ${token}` }
            });
            const data = await res.json();
            allGroups = data.groups;

            document.getElementById("test-group-select").innerHTML = allGroups
                .map(g => `<option value="${g.id}">${g.name}</option>`).join('');

            document.getElementById("ticket-recipient").innerHTML = `<option value="deanery">В Деканат</option>` +
                data.teachers.map(t => `<option value="${t.id}">Преподаватель: ${t.full_name}</option>`).join('');

            div.innerHTML = `
                <div class="flex justify-between items-center">

                    ${[0, 1, 2, 3].map(i => `
                        <div class="flex items-center gap-2">
                            <input type="radio" name="correct-${id}" value="${i}" ${i === 0 ? 'checked' : ''}>
                            <input type="text" required placeholder="Вариант ${i + 1}" class="q-opt w-full px-2.5 py-1.5 border border-slate-300 rounded-lg text-xs bg-white">

                    `).join('')}

            `;

        async function createTest(e) {
            e.preventDefault();
            const blocks = document.querySelectorAll(".q-block");
            if (!blocks.length) return alert("Добавьте хотя бы один вопрос");

            const questions = Array.from(blocks).map(b => {
                const text = b.querySelector(".q-text").value.trim();
                const options = Array.from(b.querySelectorAll(".q-opt")).map(o => o.value.trim());
                const correct_index = parseInt(b.querySelector("input[type=radio]:checked").value);
                return { text, options, correct_index };
            });

            const token = localStorage.getItem("portal_token");
            await fetch(`${API_URL}/tests`, {
                method: "POST",
                headers: { "Content-Type": "application/json", "Authorization": `Bearer ${token}` },
                body: JSON.stringify({
                    title: document.getElementById("test-title").value.trim(),
                    group_id: parseInt(document.getElementById("test-group-select").value),
                    questions
                })
            });

            document.getElementById("test-title").value = "";
            document.getElementById("test-questions-container").innerHTML = "";
            loadTests();
        }

        async function loadTests() {
            const token = localStorage.getItem("portal_token");
            const res = await fetch(`${API_URL}/tests`, {
                headers: { "Authorization": `Bearer ${token}` }
            });
            const tests = await res.json();
            window._loadedTests = tests;

            document.getElementById("tests-list").innerHTML = tests.length ? tests.map(t => `
                <div class="bg-white p-5 rounded-2xl shadow-sm border border-slate-200 flex flex-col justify-between">
                    <div>
                        <div class="flex justify-between items-center text-xs text-slate-400 mb-1">
                            <span>Группа: ${t.group_name}</span>
                            <span>Вопросов: ${t.questions.length}</span>
                        </div>
                        <h3 class="font-bold text-slate-900">${t.title}</h3>
                        <p class="text-xs text-slate-500 mt-1">Преподаватель: ${t.teacher_name}</p>
                    </div>
                    <div class="mt-4 pt-3 border-t border-slate-100 flex items-center justify-between">
                        ${currentUser.role === 'student'
                            ? (t.score !== null
                                ? `<span class="text-sm font-bold text-emerald-600">Результат: ${t.score} из ${t.max_score}</span>`
                                : `<span class="text-xs text-amber-600 font-medium">Не пройден</span>`)
                            : `<span class="text-xs text-slate-500">Сдали студентов: <b>${t.submissions}</b></span>`
                        }
                        ${currentUser.role === 'student'
                            ? `<button onclick="startTest(${t.id})" class="px-3.5 py-1.5 bg-indigo-600 hover:bg-indigo-700 text-white text-xs font-semibold rounded-xl">${t.score !== null ? 'Пересдать' : 'Начать тест'}</button>`
                            : ''
                        }
                    </div>
                </div>
            `).join('') : '<div class="text-sm text-slate-400">Назначенных тестов пока нет.</div>';
        }

        function startTest(testId) {
            activeTest = window._loadedTests.find(t => t.id === testId);
            document.getElementById("test-runner-box").classList.remove("hidden");
            document.getElementById("runner-title").textContent = activeTest.title;
            document.getElementById("runner-form").innerHTML = activeTest.questions.map((q, qIdx) => `
                <div class="p-4 bg-slate-50 rounded-xl border border-slate-200">
                    <div class="font-semibold text-sm text-slate-900 mb-2">${qIdx + 1}. ${q.text}</div>
                    <div class="space-y-1.5">
                        ${q.options.map((opt, oIdx) => `
                            <label class="flex items-center gap-2 text-sm text-slate-700 cursor-pointer">
                                <input type="radio" name="ans-${qIdx}" value="${oIdx}" required> ${opt}
                            </label>
                        `).join('')}
                    </div>
                </div>
            `).join('') + `<button type="submit" class="px-6 py-2.5 bg-emerald-600 hover:bg-emerald-700 text-white font-semibold rounded-xl text-sm">Завершить и узнать оценку</button>`;
        }

        function closeTestRunner() {
            document.getElementById("test-runner-box").classList.add("hidden");
            activeTest = null;
        }

        async function submitTestAnswers(e) {
            e.preventDefault();
            const answers = activeTest.questions.map((_, idx) => parseInt(document.querySelector(`input[name="ans-${idx}"]:checked`).value));
            const token = localStorage.getItem("portal_token");
            const res = await fetch(`${API_URL}/tests/${activeTest.id}/submit`, {
                method: "POST",
                headers: { "Content-Type": "application/json", "Authorization": `Bearer ${token}` },
                body: JSON.stringify({ answers })
            });
            const result = await res.json();
            alert(`Тест завершен! Ваш результат: ${result.score} из ${result.max_score}`);
            closeTestRunner();
            loadTests();
        }

        // --- ОБРАЩЕНИЯ (ВОПРОС-ОТВЕТ) ---
        async function createTicket(e) {
            e.preventDefault();
            const recVal = document.getElementById("ticket-recipient").value;
            const payload = {
                recipient_type: recVal === "deanery" ? "deanery" : "teacher",
                recipient_id: recVal === "deanery" ? null : parseInt(recVal),
                subject: document.getElementById("ticket-subject").value.trim(),
                question: document.getElementById("ticket-question").value.trim()
            };
            const token = localStorage.getItem("portal_token");
            await fetch(`${API_URL}/tickets`, {
                method: "POST",
                headers: { "Content-Type": "application/json", "Authorization": `Bearer ${token}` },
                body: JSON.stringify(payload)
            });
            document.getElementById("ticket-subject").value = "";
            document.getElementById("ticket-question").value = "";
            loadTickets();
        }

        async function loadTickets() {
            const token = localStorage.getItem("portal_token");
            const res = await fetch(`${API_URL}/tickets`, {
                headers: { "Authorization": `Bearer ${token}` }
            });
            const tickets = await res.json();

            document.getElementById("tickets-list").innerHTML = tickets.length ? tickets.map(tk => `
                <div class="bg-white p-5 rounded-2xl shadow-sm border border-slate-200 space-y-2">
                    <div class="flex justify-between items-center text-xs">
                        <span class="font-semibold text-slate-500">
                            От: ${tk.student_name} (${tk.group_name || 'без группы'}) → Кому: ${tk.recipient_type === 'deanery' ? 'Деканат' : tk.recipient_name}
                        </span>
                        <span class="px-2 py-0.5 rounded-lg font-semibold ${tk.status === 'answered' ? 'bg-emerald-50 text-emerald-700' : 'bg-amber-50 text-amber-700'}">
                            ${tk.status === 'answered' ? 'Отвечено' : 'Новый'}
                        </span>
                    </div>
                    <h4 class="font-bold text-slate-900 text-sm">${tk.subject}</h4>
                    <p class="text-sm text-slate-600 bg-slate-50 p-3 rounded-xl">${tk.question}</p>
                    ${tk.answer
                        ? `<div class="text-sm text-indigo-900 bg-indigo-50 p-3 rounded-xl border border-indigo-100"><b>Ответ:</b> ${tk.answer}</div>`
                        : (currentUser.role !== 'student' ? `
                            <div class="flex gap-2 pt-1">
                                <input type="text" id="ans-input-${tk.id}" placeholder="Напишите ответ студенту..." class="flex-1 px-3 py-1.5 border border-slate-300 rounded-xl text-sm">
                                <button onclick="sendTicketAnswer(${tk.id})" class="px-4 py-1.5 bg-indigo-600 hover:bg-indigo-700 text-white text-xs font-semibold rounded-xl">Ответить</button>
                            </div>
                        ` : '')
                    }
                </div>
            `).join('') : '<div class="text-sm text-slate-400">Обращений пока нет.</div>';
        }

        async function sendTicketAnswer(ticketId) {
            const answer = document.getElementById(`ans-input-${ticketId}`).value.trim();
            if (!answer) return;
            const token = localStorage.getItem("portal_token");
            await fetch(`${API_URL}/tickets/${ticketId}/answer`, {
                method: "POST",
                headers: { "Content-Type": "application/json", "Authorization": `Bearer ${token}` },
                body: JSON.stringify({ answer })
            });
            loadTickets();
        }

        // --- УПРАВЛЕНИЕ ПОЛЬЗОВАТЕЛЯМИ (ДЕКАНАТ / АДМИН) ---
        async function loadUsersList() {
            const token = localStorage.getItem("portal_token");
            const res = await fetch(`${API_URL}/users`, {
                headers: { "Authorization": `Bearer ${token}` }
            });
            if (!res.ok) return;
            const data = await res.json();

            const tbody = document.getElementById("users-table-body");
            tbody.innerHTML = data.users.map(u => {
                const isProtected = (u.id === currentUser.id) || (currentUser.role === 'deanery' && ['admin', 'deanery'].includes(u.role));
                if (isProtected) {
                    return `
                        <tr class="bg-slate-50/70">
                            <td class="py-3 px-2 font-mono text-slate-400">#${u.id}</td>
                            <td class="py-3 px-2 font-medium text-slate-700">${u.full_name}</td>
                            <td class="py-3 px-2 text-slate-400">${u.email}</td>
                            <td class="py-3 px-2"><span class="px-2.5 py-1 bg-slate-200 text-slate-700 rounded-lg text-xs font-semibold">${ROLE_NAMES[u.role]}</span></td>
                            <td class="py-3 px-2 text-slate-400 text-xs">${u.group_name || '—'}</td>
                            <td class="py-3 px-2 text-xs text-slate-400 italic">Защищено</td>
                        </tr>
                    `;
                }
                return `
                    <tr>
                        <td class="py-3 px-2 font-mono text-slate-500">#${u.id}</td>
                        <td class="py-3 px-2 font-medium">${u.full_name}</td>
                        <td class="py-3 px-2 text-slate-500">${u.email}</td>
                        <td class="py-3 px-2">
                            <select id="role-${u.id}" class="border border-slate-300 rounded-lg px-2 py-1 text-xs">
                                <option value="guest" ${u.role === 'guest' ? 'selected' : ''}>Гость</option>
                                <option value="student" ${u.role === 'student' ? 'selected' : ''}>Ученик</option>
                                <option value="teacher" ${u.role === 'teacher' ? 'selected' : ''}>Учитель</option>
                                ${currentUser.role === 'admin' ? `<option value="deanery" ${u.role === 'deanery' ? 'selected' : ''}>Деканат</option>` : ''}
                                ${currentUser.role === 'admin' ? `<option value="admin" ${u.role === 'admin' ? 'selected' : ''}>Админ</option>` : ''}
                            </select>
                        </td>
                        <td class="py-3 px-2">
                            <select id="group-${u.id}" class="border border-slate-300 rounded-lg px-2 py-1 text-xs">
                                <option value="">Без группы</option>
                                ${data.groups.map(g => `<option value="${g.id}" ${u.group_id === g.id ? 'selected' : ''}>${g.name}</option>`).join('')}
                            </select>
                        </td>
                        <td class="py-3 px-2">
                            <button onclick="saveUserRole(${u.id})" class="px-3 py-1 bg-indigo-600 hover:bg-indigo-700 text-white rounded-lg text-xs font-semibold">Сохранить</button>
                        </td>
                    </tr>
                `;
            }).join('');
        }

        async function saveUserRole(userId) {
            const token = localStorage.getItem("portal_token");
            const role = document.getElementById(`role-${userId}`).value;
            const groupVal = document.getElementById(`group-${userId}`).value;
            const group_id = groupVal ? parseInt(groupVal) : null;
            const res = await fetch(`${API_URL}/users/role`, {
                method: "POST",
                headers: { "Content-Type": "application/json", "Authorization": `Bearer ${token}` },
                body: JSON.stringify({ user_id: userId, role, group_id })
            });
            if (res.ok) {
                alert("Роль и группа обновлены!");
                loadUsersList();
            } else {
                const err = await res.json();
                alert(err.detail || "Ошибка сохранения");
            }
        }

            localStorage.removeItem("portal_token");

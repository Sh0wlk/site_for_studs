        class PrefixTree {
            constructor() { this.root = { children: {}, items: [] }; }
            insert(item, text) {
                for (const w of text.toLowerCase().split(/[\s@._,-]+/).filter(Boolean)) {
                    let node = this.root;
                    for (const ch of w) {
                        if (!node.children[ch]) node.children[ch] = { children: {}, items: [] };
                        node = node.children[ch];
                    }
                    if (!node.items.some(x => x.id === item.id)) node.items.push(item);
                }
            }
            searchByPrefix(prefix, limit = 10) {
                const clean = prefix.trim().toLowerCase();
                if (!clean) return [];
                let node = this.root;
                for (const ch of clean) {
                    if (!node.children[ch]) return [];
                    node = node.children[ch];
                }
                const res = [], seen = new Set();
                const collect = (c) => {
                    if (res.length >= limit) return;
                    for (const it of c.items) {
                        if (!seen.has(it.id)) { seen.add(it.id); res.push(it); if (res.length >= limit) return; }
                    }
                    for (const k of Object.keys(c.children)) collect(c.children[k]);
                };
                collect(node);
                return res;
            }
        }

        function compressImageFile(file, maxWidth = 300, quality = 0.8) {
            return new Promise((resolve) => {
                const reader = new FileReader();
                reader.onload = (e) => {
                    const img = new Image();
                    img.onload = () => {
                        const scale = Math.min(1, maxWidth / img.width);
                        const canvas = document.createElement("canvas");
                        canvas.width = img.width * scale;
                        canvas.height = img.height * scale;
                        canvas.getContext("2d").drawImage(img, 0, 0, canvas.width, canvas.height);
                        resolve(canvas.toDataURL("image/jpeg", quality));
                    };
                    img.src = e.target.result;
                };
                reader.readAsDataURL(file);
            });
        }

        function handleDeckSearch(q) {
            const words = q.trim().toLowerCase().split(/\s+/).filter(Boolean);
            if (!words.length) return renderDecks(allDecks);
            let matched = deckTrie.searchByPrefix(words, 50);
            for (let i = 1; i < words.length; i++) {
                const next = new Set(deckTrie.searchByPrefix(words[i], 50).map(d => d.id));
                matched = matched.filter(d => next.has(d.id));
            }
            renderDecks(matched);
        }

        function openTrainer(id) {
            const deck = allDecks.find(d => d.id === id);
            trainerQueue = [...deck.cards]; trainerTotal = deck.cards.length;
            document.getElementById("trainer-deck-title").textContent = deck.title;
            document.getElementById("modal-trainer").classList.remove("hidden");
            renderCurrentFlashcard();
        }

        function renderCurrentFlashcard() {
            document.getElementById("trainer-card-box").classList.remove("flipped");
            const learned = trainerTotal - trainerQueue.length;
            const pct = Math.round((learned / trainerTotal) * 100);
            document.getElementById("trainer-progress-bar").style.width = `${pct}%`;
            document.getElementById("trainer-stats").textContent = `Изучено: ${learned} из ${trainerTotal} (${pct}%)`;
            if (!trainerQueue.length) {
                document.getElementById("trainer-front-text").textContent = "Все карточки выучены на 100%!";
                document.getElementById("trainer-back-text").textContent = "Отличная работа!";
                document.getElementById("trainer-controls").classList.add("hidden");
                return;
            }
            document.getElementById("trainer-controls").classList.remove("hidden");
            document.getElementById("trainer-front-text").textContent = trainerQueue.front;
            document.getElementById("trainer-back-text").textContent = trainerQueue.back;
        }

        function markCardUnknown() { trainerQueue.push(trainerQueue.shift()); renderCurrentFlashcard(); }
        function markCardKnown() { trainerQueue.shift(); renderCurrentFlashcard(); }

        async function createTest(e) {
            e.preventDefault();
            const blocks = document.querySelectorAll(".q-block");
            if (!blocks.length) return alert("Добавьте вопрос");
            const questions = Array.from(blocks).map(b => ({
                text: b.querySelector(".q-text").value.trim(),
                options: Array.from(b.querySelectorAll(".q-opt")).map(o => o.value.trim()),
                correct_index: parseInt(b.querySelector("input[type=radio]:checked").value)
            }));
            await fetch(`${API_URL}/tests`, {
                method: "POST",
                headers: { "Content-Type": "application/json", "Authorization": `Bearer ${localStorage.getItem("portal_token")}` },
                body: JSON.stringify({ title: document.getElementById("test-title").value.trim(), group_id: parseInt(document.getElementById("test-group-select").value), questions })
            });
            document.getElementById("test-creator-box").classList.add("hidden");
            loadTests();
        }

        async function loadTests() {
            const res = await fetch(`${API_URL}/tests`, { headers: { "Authorization": `Bearer ${localStorage.getItem("portal_token")}` } });
            const tests = await res.json();
            window._loadedTests = tests;
            document.getElementById("tests-list").innerHTML = tests.length ? tests.map(t => `
                <div class="bg-white p-5 rounded-2xl border border-slate-200 flex flex-col justify-between">
                    <div>
                        <div class="flex justify-between text-xs text-slate-400 mb-1"><span>${t.group_name}</span><span>${t.questions.length} вопр.</span></div>
                        <h3 class="font-bold text-sm">${t.title}</h3>
                        <p class="text-xs text-slate-500 mt-1">${t.teacher_name}</p>
                    </div>
                    <div class="mt-4 pt-3 border-t flex items-center justify-between">
                        ${currentUser.role === 'student'
                            ? (t.score !== null ? `<span class="text-xs font-bold text-emerald-600">Оценка: ${t.score}/${t.max_score}</span>` : `<span class="text-xs text-amber-600">Не пройден</span>`)
                            : `<span class="text-xs text-slate-500">Сдали: <b>${t.submissions}</b></span>`
                        }
                        ${currentUser.role === 'student'
                            ? `<button onclick="startTest(${t.id})" class="px-3.5 py-1.5 bg-accent text-xs font-semibold rounded-xl">${t.score !== null ? 'Пересдать' : 'Начать'}</button>`
                            : `<button onclick="openTestReport(${t.id})" class="px-3 py-1.5 bg-slate-100 border text-xs font-semibold rounded-xl">Ведомость PDF</button>`
                        }
                    </div>
                </div>
            `).join('') : '<div class="text-sm text-slate-400">Назначенных тестов пока нет.</div>';
        }

        async function openTestReport(id) {
            const res = await fetch(`${API_URL}/tests/${id}/results`, { headers: { "Authorization": `Bearer ${localStorage.getItem("portal_token")}` } });
            const data = await res.json();
            document.getElementById("report-title").textContent = `Ведомость: ${data.title}`;
            document.getElementById("report-tbody").innerHTML = data.results.length ? data.results.map(r => `<tr><td class="p-2 border">${r.full_name}</td><td class="p-2 border">${r.group_name||'—'}</td><td class="p-2 border font-bold">${r.score}/${r.max_score}</td><td class="p-2 border">${r.date}</td></tr>`).join('') : '<tr><td colspan="4" class="p-3 text-center text-slate-400">Нет сдач</td></tr>';
            document.getElementById("test-report-box").classList.remove("hidden");
        }

        function downloadReportPDF() { html2pdf().from(document.getElementById("pdf-printable-area")).save("vedomost.pdf"); }

        function startTest(id) {
            activeTest = window._loadedTests.find(t => t.id === id);
            document.getElementById("test-runner-box").classList.remove("hidden");
            document.getElementById("runner-title").textContent = activeTest.title;
            document.getElementById("runner-form").innerHTML = activeTest.questions.map((q, qIdx) => `<div class="p-4 bg-slate-50 rounded-xl border"><div class="font-semibold text-sm mb-2">${qIdx + 1}. ${q.text}</div><div class="space-y-1.5">${q.options.map((opt, oIdx) => `<label class="flex items-center gap-2 text-xs cursor-pointer"><input type="radio" name="ans-${qIdx}" value="${oIdx}" required> ${opt}</label>`).join('')}</div></div>`).join('') + `<button type="submit" class="px-5 py-2.5 bg-emerald-600 text-white font-semibold rounded-xl text-xs">Завершить тест</button>`;
        }

        async function submitTestAnswers(e) {
            e.preventDefault();
            const answers = activeTest.questions.map((_, idx) => parseInt(document.querySelector(`input[name="ans-${idx}"]:checked`).value));
            const res = await fetch(`${API_URL}/tests/${activeTest.id}/submit`, {
                method: "POST",
                headers: { "Content-Type": "application/json", "Authorization": `Bearer ${localStorage.getItem("portal_token")}` },
                body: JSON.stringify({ answers })
            });
            const r = await res.json();
            alert(`Результат: ${r.score} из ${r.max_score}`);
            closeTestRunner(); loadTests();
        }

        async function createTicket(e) {
            e.preventDefault();
            const recVal = document.getElementById("ticket-recipient").value;
            await fetch(`${API_URL}/tickets`, {
                method: "POST",
                headers: { "Content-Type": "application/json", "Authorization": `Bearer ${localStorage.getItem("portal_token")}` },
                body: JSON.stringify({ recipient_type: recVal === "deanery" ? "deanery" : "teacher", recipient_id: recVal === "deanery" ? null : parseInt(recVal), subject: document.getElementById("ticket-subject").value.trim(), question: document.getElementById("ticket-question").value.trim() })

        async function loadTickets() {
            const res = await fetch(`${API_URL}/tickets`, { headers: { "Authorization": `Bearer ${localStorage.getItem("portal_token")}` } });
            const tickets = await res.json();
            document.getElementById("tickets-list").innerHTML = tickets.length ? tickets.map(tk => `
                <div class="bg-white p-4 rounded-2xl border border-slate-200 space-y-2">
                    <div class="flex justify-between text-xs"><span class="text-slate-500">${tk.student_name} (${tk.group_name||'—'}) → ${tk.recipient_type==='deanery'?'Деканат':tk.recipient_name}</span><span class="px-2 py-0.5 rounded-lg font-semibold ${tk.status==='answered'?'bg-emerald-50 text-emerald-700':'bg-amber-50 text-amber-700'}">${tk.status==='answered'?'Отвечено':'Новый'}</span></div>
                    <h4 class="font-bold text-sm">${tk.subject}</h4>
                    <p class="text-xs text-slate-600 bg-slate-50 p-3 rounded-xl">${tk.question}</p>
                    ${tk.answer ? `<div class="text-xs bg-indigo-50 text-indigo-900 p-3 rounded-xl"><b>Ответ:</b> ${tk.answer}</div>` : (currentUser.role !== 'student' ? `<div class="flex gap-2"><input type="text" id="ans-input-${tk.id}" placeholder="Ответ..." class="flex-1 px-3 py-1.5 border rounded-xl text-xs"><button onclick="sendTicketAnswer(${tk.id})" class="px-3.5 py-1.5 bg-accent text-xs font-semibold rounded-xl">Ответить</button></div>` : '')}

        async function sendTicketAnswer(id) {

                    "Authorization": `Bearer ${localStorage.getItem("portal_token")}`

                body: JSON.stringify({ answer })
            });

        }

        function exportAuditToExcel() {
            if (!cachedAuditLogs.length) return alert("Журнал пока пуст");
            const rows = cachedAuditLogs.map(l => ({
                "Время": l.time,
                "Инициатор": l.actor_name,
                "Событие": l.action,
                "Детали": l.details
            }));
            const ws = XLSX.utils.json_to_sheet(rows);
            ws["!cols"] = [{ wch: 20 }, { wch: 28 }, { wch: 25 }, { wch: 45 }];
            const wb = XLSX.utils.book_new();
            XLSX.utils.book_append_sheet(wb, ws, "Журнал аудита");
            XLSX.writeFile(wb, `audit_logs_${new Date().toISOString().slice(0, 10)}.xlsx`);
        }

        async function loadUsersList() {
            const res = await fetch(`${API_URL}/users`, {
                headers: { "Authorization": `Bearer ${localStorage.getItem("portal_token")}` }
            });
            if (!res.ok) return;
            const data = await res.json();
            cachedUsers = data.users;
            cachedGroups = data.groups;
            userTrie = new PrefixTree();
            cachedUsers.forEach(u => userTrie.insert(u, `${u.full_name} ${u.email}`));
            renderUsersTable(cachedUsers);
        }

        function handleUserSearch(q) {
            const box = document.getElementById("search-suggestions");
            if (!q.trim()) {
                box.classList.add("hidden");
                return renderUsersTable(cachedUsers);
            }
            const matches = userTrie.searchByPrefix(q, 5);
            box.innerHTML = matches.length ? matches.map(u => `
                <button type="button" onclick="selectSearchUser(${u.id}, '${u.full_name.replace(/'/g, "\\'")}')"
                        class="w-full text-left px-4 py-2 hover:bg-slate-50 flex justify-between border-b text-xs">
                    <span><b>${u.full_name}</b> (${u.email})</span>
                    <span class="text-accent">${ROLE_NAMES[u.role]}</span>
                </button>
            `).join('') : `<div class="px-4 py-2 text-xs text-slate-400">Ничего не найдено</div>`;
            box.classList.remove("hidden");
            renderUsersTable(matches);
        }

        function selectSearchUser(id, name) {
            document.getElementById("user-search-input").value = name;
            document.getElementById("search-suggestions").classList.add("hidden");
            renderUsersTable(cachedUsers.filter(u => u.id === id));
        }

        function renderUsersTable(list) {
            document.getElementById("users-table-body").innerHTML = list.map(u => {
                const isProtected = (u.id === currentUser.id) || (currentUser.role === 'deanery' && ['admin', 'deanery'].includes(u.role));
                if (isProtected) {
                    return `
                        <tr class="bg-slate-50/70">
                            <td class="py-2.5 px-2 font-mono text-slate-400">#${u.id}</td>
                            <td class="py-2.5 px-2 font-medium flex items-center gap-2">
                                <img src="${resolveAvatar(u)}" class="w-7 h-7 rounded-lg object-cover">${u.full_name}
                            </td>
                            <td class="py-2.5 px-2 text-slate-400">${u.email}</td>
                            <td class="py-2.5 px-2"><span class="px-2 py-0.5 bg-slate-200 rounded text-xs">${ROLE_NAMES[u.role]}</span></td>
                            <td class="py-2.5 px-2">${u.group_name || '—'}</td>
                            <td class="py-2.5 px-2 text-slate-400 italic">Защищено</td>
                        </tr>`;
                }
                return `
                    <tr>
                        <td class="py-2.5 px-2 font-mono text-slate-500">#${u.id}</td>
                        <td class="py-2.5 px-2 font-medium flex items-center gap-2">
                            <img src="${resolveAvatar(u)}" class="w-7 h-7 rounded-lg object-cover">${u.full_name}
                        </td>
                        <td class="py-2.5 px-2 text-slate-500">${u.email}</td>
                        <td class="py-2.5 px-2">
                            <select id="role-${u.id}" class="border rounded-lg px-2 py-1 text-xs">
                                <option value="guest" ${u.role === 'guest' ? 'selected' : ''}>Гость</option>
                                <option value="student" ${u.role === 'student' ? 'selected' : ''}>Ученик</option>
                                <option value="teacher" ${u.role === 'teacher' ? 'selected' : ''}>Учитель</option>
                                ${currentUser.role === 'admin' ? `<option value="deanery" ${u.role === 'deanery' ? 'selected' : ''}>Деканат</option>` : ''}
                                ${currentUser.role === 'admin' ? `<option value="admin" ${u.role === 'admin' ? 'selected' : ''}>Админ</option>` : ''}
                            </select>
                        </td>
                        <td class="py-2.5 px-2">
                            <select id="group-${u.id}" class="border rounded-lg px-2 py-1 text-xs">
                                <option value="">Без группы</option>
                                ${cachedGroups.map(g => `<option value="${g.id}" ${u.group_id === g.id ? 'selected' : ''}>${g.name}</option>`).join('')}
                            </select>
                        </td>
                        <td class="py-2.5 px-2">
                            <button onclick="saveUserRole(${u.id})" class="px-3 py-1 bg-accent rounded-lg text-xs font-semibold">Сохранить</button>
                        </td>
                    </tr>`;
            }).join('');
        }

        async function saveUserRole(userId) {
            const role = document.getElementById(`role-${userId}`).value;
            const gVal = document.getElementById(`group-${userId}`).value;
            const res = await fetch(`${API_URL}/users/role`, {
                method: "POST",
                headers: {
                    "Content-Type": "application/json",
                    "Authorization": `Bearer ${localStorage.getItem("portal_token")}`
                },
                body: JSON.stringify({ user_id: userId, role, group_id: gVal ? parseInt(gVal) : null })
            });
            if (res.ok) {
                alert("Сохранено!");
                loadUsersList();
            }
        }
